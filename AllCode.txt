function [sh, ubl, PBL , PobMax] = parametrDN(xxx,mas,uroven)
% Ширина луча в градусах из координаты Х, ДН в дБ, уровень в дБ


% если не соразмерны массивы
if length(xxx) ~= length(mas) 
    sh = 0; ubl = 0; PBL_Ldb=0; PBL_Rdb;
    warning('beam_width: Uncorrect data'); return
end


% ШИРИНА ЛУЧА
maxDN=0;
[~,max_c]=find(mas==maxDN);
cut=(sign(mas - uroven)+1)/2;
non=nonzeros(xxx.*cut); %остались координаты x, где Sdbx>-3
if (asind(xxx(1,max_c)))>30
    non(find(non<0))=[];
elseif (asind(xxx(1,max_c)))<-30
    non(find(non>0))=[];
end;
sh=180/pi*(asin(max(non))-asin(min(non))); %ширина сумм ДН по уровню -3 дБ в сечении плоскостью x0z
num = length(xxx);

% УБЛ
try
    for t=1:num              %меньший индекс - первый максимум
        if cut(t)==1
            i1=t; break     
        end
    end
    for t=num:-1:1           %больший индекс - второй максимум
        if cut(t)==1
            i2=t; break     
        end
    end    
    for t=i2:num %находим индексы соответствующие ГЛ по смене знака производной - #todo, пока неправда
        if mas(t+1)-mas(t)>0, t0=t; break
        end
    end

    for t=i1:-1:2
        if mas(t-1)-mas(t)>0, t2=t; break
        end
        t2=t;
    end
    % ubl= 0; %приходится применять для ШИРа

    ubl = max([max(mas(t0:num)),max(mas(1:t2))]); % максимальное значение ДН в оставшемся диапазоне - УБЛ
catch    
    ubl= 0; disp('УБЛ не посчитан');
end

PobMax=-990; PBL=-999;

% ПБЛ
 th1=asind(xxx(1,max_c))-sh*10; 
 th1=round(th1,2);
 kord1=find(round(asind(xxx),2)==th1);
 th2=asind(xxx(1,max_c))+sh*10; 
 th2=round(th2,2);
 kord2=find(round(asind(xxx),2)==th2);
 PBL_Ldb=-999; PBL_Rdb=-999; PBL_Lgr=-999; PBL_Rgr=-999;
 k=0;
 for i=flip(kord1:max_c)
   if k==0
    if mas(i)<maxDN && mas(i)>PBL_Ldb && (mas(i-1)<mas(i) && mas(i)>mas(i+1))
        PBL_Ldb=mas(i);
        PBL_Lgr=asind(xxx(i));
        kord_L=i;
        k=1;
    end;
   end;
 end;
 k=0;
 for i=max_c:kord2
   if k==0
    if mas(i)<maxDN && mas(i)>PBL_Rdb && (mas(i-1)<mas(i) && mas(i)>mas(i+1))
     PBL_Rdb=mas(i);
     PBL_Rgr=asind(xxx(i));
     kord_R=i;
     k=1;
    end;
   end;
 end;
 PBL_L=[round(PBL_Ldb,2),round(PBL_Lgr,2)];
 PBL_R=[round(PBL_Rdb,2),round(PBL_Rgr,2)];
 PBL=[PBL_L;PBL_R];
 
 % побочный максимум
 m=size(mas); m=m(2);
 PobMax_Ldb=-999; PobMax_Rdb=-999;
 PobMax_Lgr=-999; PobMax_Rgr=-999;
 PobMax_Lind=-999; PobMax_Rind=999;
 for i=flip(2:kord_L)
     if  mas(i-1)<mas(i) && mas(i)>mas(i+1) && mas(i)>PobMax_Ldb && i~=kord_L
         PobMax_Ldb=mas(i);
         PobMax_Lgr=asind(xxx(i));
         PobMax_Lind=i;
     end;
 end;
  for i=kord_R:(m-1)
     if  mas(i-1)<mas(i) && mas(i)>mas(i+1) && mas(i)>PobMax_Rdb && i~=kord_R
         PobMax_Rdb=mas(i);
         PobMax_Rgr=asind(xxx(i));
         PobMax_Rind=i;
     end;
 end;
 PobMax_L=[round(PobMax_Ldb,2),round(PobMax_Lgr,2),PobMax_Lind];
 PobMax_R=[round(PobMax_Rdb,2),round(PobMax_Rgr,2),PobMax_Rind];
 PobMax=[PobMax_L; PobMax_R ];
 clear kord1 kord2 th1 th2
end

function [ DS_dB  ] = DNscan( xx,yy, whichDS )
%UNTITLED Загрузка Диаграммы Сканирования элемента
%   Входные параметры:
%   whichDS - выбор ДН элемента (DSxoz - плоскость XOZ; 
%             DSyoz - плоскость YOZ; DS3d - 3D ДН; DSalpha - сечение
%             плоскостью альфа).
%   ThMin - минимальное значение Th в диапозоне сканирования ДН.
%   ThMax - максимальное значение Th в диапозоне сканирования ДН.
%   stepTh - шаг диапозона сканирования ДН.
%   file - подгрузка файла ДН (Пример: 'G:\папка\DS.txt').
% 
%   Выходные параметры:
%   DS_dB - ДН в дБ.

global lit

%% Средняя лит
if lit == 39 || lit == 41 || lit == 53 || lit == 63 || lit == 51
    importDS = importdata('F:\ДН_моделинг\v.2.1-27.08.2024\DS_s.txt');
   
    % whichDS='DS3d';    %DSxoz DSyoz DS3d DSalpha

    m=size(xx); m=m(2);
    if strcmp(whichDS, 'DSxoz') == 1
        xx=round(xx,2);
        stepTh=abs(xx(1)-xx(2));
        [row, col]=find(importDS(:,1)==0); 
%     DsR_gr=importDS(row,2); [m,p]=size(DsR_gr); DsL_gr=-DsR_gr(flip(2:m));
%     Ds_gr=cat(1,DsL_gr,DsR_gr);
        DsR_dB=importDS(row,3); [u]=size(DsR_dB); DsL_dB=DsR_dB(flip(2:u(1)));
        DSs_dB=cat(1,DsL_dB,DsR_dB); 
        DSs_dB=transpose(DSs_dB); rtw=size(DSs_dB); rtw=rtw(2);
        DS_int=interp1(DSs_dB,(1:0.1:rtw), 'spline');
        Gr=-89:0.01:89; [n]=size(DS_int); n=n(2);
        DS_dB=ones(1,m);
        for j=1:m
         if xx(j)<Gr(1) DS_dB(j)=DS_int(1); end;
         if xx(j)>Gr(n) DS_dB(j)=DS_int(n); end;
         for i=1:n
         if (xx(j)>=(Gr(i)-stepTh/10)) && (xx(j)<=(Gr(i)+stepTh/10)) 
           DS_dB(j)=DS_int(i); end;
         end;
        end;
%     figure
%     plot(Ds_gr, Ds_dB);   
   
    elseif strcmp(whichDS, 'DSyoz') == 1
        xx=round(xx,2);
        stepTh=abs(xx(1)-xx(2));
        [row, col]=find(importDS(:,1)==90); 
%     DsR_gr=importDS(row,2); [m,p]=size(DsR_gr); DsL_gr=-DsR_gr(flip(2:m));
%     Ds_gr=cat(1,DsL_gr,DsR_gr);
        DsR_dB=importDS(row,3); [u]=size(DsR_dB); DsL_dB=DsR_dB(flip(2:u(1)));
        DSs_dB=cat(1,DsL_dB,DsR_dB);
        DSs_dB=transpose(DSs_dB); rtw=size(DSs_dB); rtw=rtw(2);
        DS_int=interp1(DSs_dB,(1:0.1:rtw), 'spline');
        Gr=-89:0.01:89; [n]=size(Gr); n=n(2);
        DS_dB=ones(1,m);
        for j=1:m
           if xx(j)<Gr(1) DS_dB(j)=DS_int(1); end;
           if xx(j)>Gr(n) DS_dB(j)=DS_int(n); end;
           for i=1:n
            if (xx(j)>=(Gr(i)-stepTh/10)) && (xx(j)<=(Gr(i)+stepTh/10)) 
             DS_dB(j)=DS_int(i); end;
           end;
        end;
%     figure
%     plot(Ds_gr, Ds_dB);
    
    elseif strcmp(whichDS, 'DSalpha') == 1
       [row, col]=find(importDS(:,1)==alpha); 
%     DsR_gr=importDS(row,2); [m,p]=size(DsR_gr); DsL_gr=-DsR_gr(flip(2:m));
%     Ds_gr=cat(1,DsL_gr,DsR_gr);
       DsR_dB=importDS(row,3); [u]=size(DsR_dB); DsL_gr=DsR_dB(flip(2:u(1)));
       DSs_dB=cat(1,DsL_gr,DsR_dB);
       DSs_dB=transpose(DSs_dB);
       DS_int=interp1(DSs_dB,(1:0.1:rtw), 'spline');
       Gr=-89:0.01:89; [n]=size(Gr); n=n(2)
       DS_dB=ones(1,m);
       for j=1:m
         if xx(j)<Gr(1) DS_dB(j)=DS_int(1); end;
         if xx(j)>Gr(n) DS_dB(j)=DS_int(n); end;
         for i=1:n
           if (xx(j)>=(Gr(i)-stepTh/10)) && (xx(j)<=(Gr(i)+stepTh/10)) 
            DS_dB(j)=DS_int(i); end;
         end;
       end;

    elseif strcmp(whichDS, 'DS3d') == 1
      f = repmat(0:90,891,1);
      t = repmat((0:0.1:89)',1,91);
      x = asind(sind(t).*sind(f));
      y = asind(sind(t).*cosd(f));
      k=1;
      for i=flip(1:91);
        s=0;
        for j=k:(k+890)
            s=s+1;
            ds(s,i)=importDS(j,3);
            if mod(j,891)==0 
                k=k+891;
                s=0;
            end
        end
      end
      Bds=[flip(ds); ds]; % 1 и 4 дб
      Cds=[flip(Bds,2) Bds]; % все кв дб
      BXds=[flip(x); x];  % 1 и 4 по х
      CXds=[-flip(BXds,2) BXds]; %все кв по х
      BYds=[-flip(y); y];  % 1 и 4 по у
      CYds=[-flip(BYds,2),BYds]; % все по у
      DS_dB = griddata(CXds,CYds,Cds,yy,xx);    
    % оси уу и хх поменяты местами в связи с тем что ДС почему-то
    % поворачивается на 90 градусов, такой порядок (уу, хх) все чинит
    end;

%% Нижняя лит
elseif lit == 27 || lit == 29 || lit == 1
%     importDS=flip(importdata('G:\ДН_моделинг\v.2.1-27.08.2024\DS_s.txt'),1);
    importDS=importdata('F:\ДН_моделинг\v.2.1-27.08.2024\DS_n.txt');
    % whichDS='DS3d';    %DSxoz DSyoz DS3d DSalpha

    m=size(xx); m=m(2);
    if strcmp(whichDS, 'DSxoz') == 1
        xx=round(xx,2);

      BYds=[-flip(y); y];  % 1 и 4 по у
      CYds=[-flip(BYds,2),BYds]; % все по у

    % поворачивается на 90 градусов, такой порядок (уу, хх) все чинит
    end;

function [ Ykord ] = KordSetkaOY( gekso, Nx, Ny, dy, Nybalk, yb  )
%KordSetkaOY - Построение координатной сетки (массив Ykord) по оси OY
%   gekso - Определение гексогональности сетки
%   Nx - Кол-во элементов по оси ОХ
%   Ny - Кол-во элементов по оси ОУ
%   dy - шаг по оси ОХ
%   Nybalk - Кол-во вертикальных балок
%   yb - Ширина вертикальной балки
if mod(Ny,2)==0
    k=0;
%     Нет балок
    if (Nybalk==0)||(yb==0)
        for n=1:Nx
            for i=flip(1:Ny/2)
                if gekso==0 c=1; else c=0; end;
                if gekso==1 g=1; else g=0; end;
                if (gekso==1)&&(mod(n,2)==1) a=1; else a=0; end;
                Ykord(i,n)=-a*dy/2-k*dy-g*dy/4-c*dy/2;
                k=k+1;
            end;
            k=0; a=0;
            for i=(Ny/2+1):Ny
                if gekso==0 c=1; else c=0; end;
                if gekso==1 g=1; else g=0; end;
                if (gekso==1)&&(mod(n,2)==0) a=1; else a=0; end;
                Ykord(i,n)=a*dy/2+k*dy+g*dy/4+c*dy/2;
                k=k+1;
            end;
            k=0;
        end;
    end;
    
%     Нечетное кол-во балок
    if ((yb~=0)||(Nybalk~=0))&(mod(Nybalk,2)==1)
        Nsec=Nybalk+1;
        Nysec=Ny/Nsec;
        if mod(Nysec,1)==0
            for n=1:Nx
                k=0; kb=-1;
                for i=flip(1:Ny/2)
                    if gekso==0 c=1; else c=0; end;
                    if gekso==1 g=1; else g=0; end;
                    if (gekso==1)&&(mod(n,2)==1) a=1; else a=0; end; 
                    if mod(i,Nysec)==0 kb=kb+1; end; 
                    Ykord(i,n)=-yb/2-kb*yb-a*dy/2-k*dy-g*dy/4-c*dy/2;
                    k=k+1;
                end;
                kb=-1; k=0;
                for i=(Ny/2+1):Ny
                    if gekso==0 c=1; else c=0; end;
                    if gekso==1 g=1; else g=0; end;
                    if (gekso==1)&&(mod(n,2)==0) a=1; else a=0; end; 
                    if mod(i,Nysec)==1 kb=kb+1; end; 
                    Ykord(i,n)=yb/2+kb*yb+a*dy/2+k*dy+g*dy/4+c*dy/2;
                    k=k+1;
                end;
            end;
        else Ykord='Ошибка 4. Некорректное соотношение балок и элементов по оси ОУ.'
        end;
    end;
    
%     Четное кол-во балок
    if ((yb~=0)||(Nybalk~=0))&(mod(Nybalk,2)==0)
        Nsec=Nybalk+1;
        Nysec=Ny/Nsec;
        if mod(Nysec,1)==0
            for n=1:Nx
                k=0; kb=0;
                for i=flip(1:Ny/2)
                    if gekso==0 c=1; else c=0; end;
                    if gekso==1 g=1; else g=0; end;
                    if (mod(n,2)==1) && (gekso==1) a=1; else a=0; end;
                    if mod(i,Nysec)==0 kb=kb+1; end;
                    Ykord(i,n)=-kb*yb-a*dy/2-k*dy-g*dy/4-c*dy/2;  
                    k=k+1;
                end;
                k=0; kb=0;
                for i=(Ny/2+1):Ny
                    if gekso==0 c=1; else c=0; end;
                    if gekso==1 g=1; else g=0; end;
                    if (mod(n,2)==0) && (gekso==1) a=1; else a=0; end;
                    if mod(i,Nysec)==1 kb=kb+1; end;
                    Ykord(i,n)=kb*yb+a*dy/2+k*dy+g*dy/4+c*dy/2;
                    k=k+1;
                end;
            end;
        else Ykord='Ошибка 4. Некорректное соотношение балок и элементов по оси ОУ.'
        end;
    end; end;
clear k kb Nsec Nxsec x1 x2

end

function [sh, ubl, PBL , PobMax] = parametrDN(xxx,mas,uroven)
% Ширина луча в градусах из координаты Х, ДН в дБ, уровень в дБ


% если не соразмерны массивы
if length(xxx) ~= length(mas) 
    sh = 0; ubl = 0; PBL_Ldb=0; PBL_Rdb;
    warning('beam_width: Uncorrect data'); return
end


% ШИРИНА ЛУЧА
maxDN=0;
[~,max_c]=find(mas==maxDN);
cut=(sign(mas - uroven)+1)/2;
non=nonzeros(xxx.*cut); %остались координаты x, где Sdbx>-3
if (asind(xxx(1,max_c)))>30
    non(find(non<0))=[];
elseif (asind(xxx(1,max_c)))<-30
    non(find(non>0))=[];
end;
sh=180/pi*(asin(max(non))-asin(min(non))); %ширина сумм ДН по уровню -3 дБ в сечении плоскостью x0z
num = length(xxx);

% УБЛ
try
    for t=1:num              %меньший индекс - первый максимум
        if cut(t)==1
            i1=t; break     
        end
    end
    for t=num:-1:1           %больший индекс - второй максимум
        if cut(t)==1
            i2=t; break     
        end
    end    
    for t=i2:num %находим индексы соответствующие ГЛ по смене знака производной - #todo, пока неправда
        if mas(t+1)-mas(t)>0, t0=t; break
        end
    end

    for t=i1:-1:2
        if mas(t-1)-mas(t)>0, t2=t; break
        end
        t2=t;
    end
    % ubl= 0; %приходится применять для ШИРа

    ubl = max([max(mas(t0:num)),max(mas(1:t2))]); % максимальное значение ДН в оставшемся диапазоне - УБЛ
catch    
    ubl= 0; disp('УБЛ не посчитан');
end

PobMax=-990; PBL=-999;

% ПБЛ
 th1=asind(xxx(1,max_c))-sh*10; 
 th1=round(th1,2);
 kord1=find(round(asind(xxx),2)==th1);
 th2=asind(xxx(1,max_c))+sh*10; 
 th2=round(th2,2);
 kord2=find(round(asind(xxx),2)==th2);
 PBL_Ldb=-999; PBL_Rdb=-999; PBL_Lgr=-999; PBL_Rgr=-999;
 k=0;
 for i=flip(kord1:max_c)
   if k==0
    if mas(i)<maxDN && mas(i)>PBL_Ldb && (mas(i-1)<mas(i) && mas(i)>mas(i+1))
        PBL_Ldb=mas(i);
        PBL_Lgr=asind(xxx(i));
        kord_L=i;
        k=1;
    end;
   end;
 end;
 k=0;
 for i=max_c:kord2
   if k==0
    if mas(i)<maxDN && mas(i)>PBL_Rdb && (mas(i-1)<mas(i) && mas(i)>mas(i+1))
     PBL_Rdb=mas(i);
     PBL_Rgr=asind(xxx(i));
     kord_R=i;
     k=1;
    end;
   end;
 end;
 PBL_L=[round(PBL_Ldb,2),round(PBL_Lgr,2)];
 PBL_R=[round(PBL_Rdb,2),round(PBL_Rgr,2)];
 PBL=[PBL_L;PBL_R];
 
 % побочный максимум
 m=size(mas); m=m(2);
 PobMax_Ldb=-999; PobMax_Rdb=-999;
 PobMax_Lgr=-999; PobMax_Rgr=-999;
 PobMax_Lind=-999; PobMax_Rind=999;
 for i=flip(2:kord_L)
     if  mas(i-1)<mas(i) && mas(i)>mas(i+1) && mas(i)>PobMax_Ldb && i~=kord_L
         PobMax_Ldb=mas(i);
         PobMax_Lgr=asind(xxx(i));
         PobMax_Lind=i;
     end;
 end;
  for i=kord_R:(m-1)
     if  mas(i-1)<mas(i) && mas(i)>mas(i+1) && mas(i)>PobMax_Rdb && i~=kord_R
         PobMax_Rdb=mas(i);
         PobMax_Rgr=asind(xxx(i));
         PobMax_Rind=i;
     end;
 end;
 PobMax_L=[round(PobMax_Ldb,2),round(PobMax_Lgr,2),PobMax_Lind];
 PobMax_R=[round(PobMax_Rdb,2),round(PobMax_Rgr,2),PobMax_Rind];
 PobMax=[PobMax_L; PobMax_R ];
 clear kord1 kord2 th1 th2
end
